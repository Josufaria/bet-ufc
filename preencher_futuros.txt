function preencherFuturosCompletos() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const ui = SpreadsheetApp.getUi();
  let abasCorrigidas = 0;
  
  for (const ev of DADOS_FUTUROS) {
    let sheet = ss.getSheetByName(ev.short_name);
    if (!sheet) {
       // Se não existir, tenta achar pelo nome completo
       sheet = ss.getSheetByName(ev.name);
       if (!sheet) {
           sheet = ss.insertSheet(ev.short_name);
       }
    }
    
    // Limpar aba
    sheet.clear();
    
    // 1. CABEÇALHO DO EVENTO
    sheet.getRange('A1').setValue(ev.date);
    
    // 2. CARD DE LUTAS
    sheet.getRange(3, 1, 1, 5).setValues([['CARD DE LUTAS', '', '', '', '']]);
    sheet.getRange(4, 1, 1, 5).setValues([['Lutador V', 'Odd V', '', 'Odd A', 'Lutador A']]);
    
    let rowIdx = 5;
    for (const f of ev.fights) {
      sheet.getRange(rowIdx, 1, 1, 5).setValues([[f.f1, f.oV, 'vs', f.oA, f.f2]]);
      rowIdx++;
    }
    
    // Como é evento FUTURO, deixamos RESULTADO REAL VAZIO ou não colocamos
    // Mas o PunterMind analisa o EV+ antes da luta acontecer!
    
    // 3. PUNTERMIND ANALISE PROFUNDA
    const pmStart = rowIdx + 2;
    sheet.getRange(pmStart, 1, 1, 7).setValues([['', 'PUNTERMIND ANALISE PROFUNDA', '', '', '', '', '']]);
    sheet.getRange(pmStart + 1, 1, 1, 7).setValues([['Luta', 'Fisico/Matchup', 'Prob Real', 'EV+', 'Pick Recomendada', 'Metodo', 'Nota / Justificativa']]);
    
    let pmRow = pmStart + 2;
    for (const f of ev.fights) {
      sheet.getRange(pmRow, 1, 1, 7).setValues([[f.f1 + ' vs ' + f.f2, f.fisico, f.probReal, f.ev, f.pick, f.met, f.nota]]);
      pmRow++;
    }
    
    // 4. APOSTAS SUGERIDAS
    const apStart = pmRow + 2;
    sheet.getRange(apStart, 1, 1, 5).setValues([['', 'MELHORES APOSTAS DO EVENTO', '', '', '']]);
    sheet.getRange(apStart + 1, 1, 1, 5).setValues([['Tipo', 'Selecao', 'Odd Media', 'Confianca', 'Justificativa']]);
    if (ev.fights.length >= 2) {
       const fA = ev.fights[0]; const fB = ev.fights[1];
       sheet.getRange(apStart + 2, 1, 1, 5).setValues([['Simples (Melhor Pick)', fA.pick, fA.oV, 'Alta', 'Melhor aposta EV+ do card principal.']]);
       sheet.getRange(apStart + 3, 1, 1, 5).setValues([['Dupla Segura', fA.pick + ' + ' + fB.pick, (parseFloat(fA.oV) * parseFloat(fB.oV)).toFixed(2), 'Media', 'Maior favoritismo consolidado.']]);
    }
    
    abasCorrigidas++;
  }
  
  ui.alert('Pronto!', 'Foram inseridos os cards completos e as estatísticas em ' + abasCorrigidas + ' eventos futuros!', ui.ButtonSet.OK);
}
