function criarEventosFuturos() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let template = ss.getSheetByName('TEMPLATE_CARD');
  
  // Se o usuário deletou o TEMPLATE_CARD, usamos o primeiro evento disponivel como clone
  if (!template) {
     const sheets = ss.getSheets();
     template = sheets.find(s => s.getName().startsWith('FN ') || s.getName().startsWith('UFC '));
     if (!template) {
       SpreadsheetApp.getUi().alert('Não foi possível achar um template para copiar!');
       return;
     }
  }
  
  let abasCriadas = 0;
  
  for (const ev of EVENTOS_FUTUROS) {
    // Formatar nome
    let shortName = ev.name.replace('UFC Fight Night: ', 'FN ').replace('vs.', 'vs');
    if (shortName.startsWith('UFC ') && shortName.includes(':')) {
      shortName = shortName.split(':')[0];
    }
    shortName = shortName.substring(0, 30);
    
    let sheet = ss.getSheetByName(shortName);
    if (!sheet) {
      sheet = template.copyTo(ss);
      sheet.setName(shortName);
      abasCriadas++;
      
      // Limpar o conteudo da aba clonada (Mudar Nome e Data)
      sheet.getRange('A1').setValue(ev.date);
      
      // Limpar os lutadores antigos e colocar a nova luta principal
      sheet.getRange('A5:E20').clearContent();
      
      let rowIdx = 5;
      for (const f of ev.fights) {
        sheet.getRange(rowIdx, 1, 1, 5).setValues([[f.f1, '', '', '', f.f2]]);
        rowIdx++;
      }
      
      // Limpar a seção do puntermind
      sheet.getRange('A24:G40').clearContent();
    }
  }
  
  SpreadsheetApp.getUi().alert('Futuros Adicionados!', 'Foram criadas ' + abasCriadas + ' abas para eventos futuros já com as lutas principais!', SpreadsheetApp.getUi().ButtonSet.OK);
}
