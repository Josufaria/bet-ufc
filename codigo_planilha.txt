function preencherTudo() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const ui = SpreadsheetApp.getUi();
  let abasAtualizadas = 0;
  
  for (const ev of DADOS_COMPLETOS) {
    const sheet = ss.getSheetByName(ev.short_name);
    if (!sheet) {
        Logger.log("Nao encontrou a aba: " + ev.short_name);
        continue;
    }
    
    let rowIdx = 5;
    for (const f of ev.fights) {
      sheet.getRange(rowIdx, 2).setValue(f.oV);
      sheet.getRange(rowIdx, 4).setValue(f.oA);
      rowIdx++;
    }
    
    const resultStart = rowIdx + ev.fights.length + 3;
    sheet.getRange(resultStart, 1, 1, 7).setValues([['', 'PUNTERMIND ANALISE PROFUNDA', '', '', '', '', '']]);
    sheet.getRange(resultStart + 1, 1, 1, 7).setValues([['Luta', 'Fisico/Matchup', 'Prob Real', 'EV+', 'Pick Recomendada', 'Metodo', 'Nota / Justificativa']]);
    
    let analiseRow = resultStart + 2;
    for (const f of ev.fights) {
      sheet.getRange(analiseRow, 1, 1, 7).setValues([[f.f1 + ' vs ' + f.f2, f.fisico, f.probReal, f.ev, f.pick, f.met, f.nota]]);
      analiseRow++;
    }
    
    const apostaStart = analiseRow + 1;
    sheet.getRange(apostaStart, 1, 1, 5).setValues([['', 'MELHORES APOSTAS DO EVENTO', '', '', '']]);
    sheet.getRange(apostaStart + 1, 1, 1, 5).setValues([['Tipo', 'Selecao', 'Odd Media', 'Confianca', 'Justificativa']]);
    if (ev.fights.length >= 2) {
       const fA = ev.fights[0]; const fB = ev.fights[1];
       sheet.getRange(apostaStart + 2, 1, 1, 5).setValues([['Simples (Melhor Pick)', fA.pick, fA.oV, 'Alta', 'Vantagem clara.']]);
       sheet.getRange(apostaStart + 3, 1, 1, 5).setValues([['Dupla Segura', fA.pick + ' + ' + fB.pick, (parseFloat(fA.oV) * parseFloat(fB.oV)).toFixed(2), 'Media', 'Maior favoritismo do card.']]);
    }
    abasAtualizadas++;
  }
  
  const sheets = ss.getSheets();
  let sheetLut = null;
  for (const s of sheets) {
    if (s.getName().indexOf('Lutadores') !== -1) {
      sheetLut = s;
      break;
    }
  }
  
  if (sheetLut) {
    const data = sheetLut.getDataRange().getValues();
    const cadastrados = new Set(data.map(row => String(row[0]).trim().toLowerCase()));
    
    let adic = 0;
    const lastRow = sheetLut.getLastRow();
    const rowsToAdd = [];
    for (const lut of LUTADORES_NOVOS) {
      if (!cadastrados.has(lut.nome.toLowerCase())) {
        rowsToAdd.push([lut.nome, lut.estilo, '', '', '', '', '', '']);
        cadastrados.add(lut.nome.toLowerCase());
        adic++;
      }
    }
    if (rowsToAdd.length > 0) sheetLut.getRange(lastRow + 1, 1, rowsToAdd.length, 8).setValues(rowsToAdd);
  }
  
  ui.alert('Site Certinho!', 'Foram preenchidas as Odds e Analises em ' + abasAtualizadas + ' abas, e lutadores cadastrados!', ui.ButtonSet.OK);
}
